03: // ===== 文件：backend/migrations/096_foreign_key_index_coverage.sql =====
04: -- PostgreSQL does not create indexes on referencing columns automatically.
05: -- Add a leading index for every foreign key that remains uncovered.
06:
07: DO $$
08: DECLARE
09:   fk record;
10:   index_name text;
11: BEGIN
12:   FOR fk IN
13:     SELECT
14:       c.conrelid,
15:       c.conrelid::regclass AS table_name,
16:       c.conname,
17:       c.conkey,
18:       string_agg(quote_ident(a.attname), ', ' ORDER BY key_column.ordinality) AS columns_sql
19:     FROM pg_constraint c
20:     JOIN pg_class relation ON relation.oid = c.conrelid
21:     JOIN pg_namespace namespace ON namespace.oid = relation.relnamespace
22:     CROSS JOIN LATERAL unnest(c.conkey) WITH ORDINALITY AS key_column(attnum, ordinality)
23:     JOIN pg_attribute a
24:       ON a.attrelid = c.conrelid
25:      AND a.attnum = key_column.attnum
26:     WHERE c.contype = 'f'
27:       AND namespace.nspname = current_schema()
28:       AND NOT EXISTS (
29:         SELECT 1
30:         FROM pg_index i
31:         WHERE i.indrelid = c.conrelid
32:           AND i.indisvalid
33:           AND (
34:             SELECT array_agg(index_column.attnum ORDER BY index_column.ordinality)
35:             FROM unnest(i.indkey) WITH ORDINALITY AS index_column(attnum, ordinality)
36:             WHERE index_column.ordinality <= cardinality(c.conkey)
37:           ) = c.conkey
38:       )
39:     GROUP BY c.conrelid, c.conname, c.conkey
40:   LOOP
41:     index_name :=
42:       left(replace(fk.table_name::text, '.', '_') || '_' || fk.conname, 54)
43:       || '_'
44:       || substr(md5(fk.conrelid::text || ':' || fk.conname), 1, 8);
45:     EXECUTE format(
46:       'CREATE INDEX IF NOT EXISTS %I ON %s (%s)',
47:       index_name,
48:       fk.table_name,
49:       fk.columns_sql
50:     );
