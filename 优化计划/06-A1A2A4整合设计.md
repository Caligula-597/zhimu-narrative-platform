### 4.3 createStore 设计

```js
// src/state/create-store.js
/**
 * Lightweight store factory — no dependencies, no framework.
 * Inspired by play/host state pattern, upgraded with subscribe + batch.
 *
 * Usage:
 *   const store = createStore({ count: 0 });
 *   store.get().count;              // 0
 *   store.set({ count: 1 });        // notify listeners
 *   store.subscribe((next, prev) => {
 *     if (next.count !== prev.count) render();
 *   });
 */
export function createStore(initial) {
  let state = { ...initial };
  const listeners = new Set();
  let batching = false;
  let pending = null;

  function notify(next, prev) {
    for (const fn of listeners) {
      try { fn(next, prev); } catch (e) { console.error("[store listener]", e); }
    }
  }

  return {
    get: () => state,
    set(patch) {
      const prev = state;
      const next = { ...state, ...patch };
      state = next;
      if (batching) {
        pending = pending ? { ...pending, ...patch } : patch;
      } else {
        notify(next, prev);
      }
      return next;
    },
    /** Batch multiple set() calls into one notify. */
    batch(fn) {
      batching = true;
      const prev = state;
      try { fn(); }
      finally {
        batching = false;
        if (pending) {
          const next = state;
          notify(next, prev);
          pending = null;
        }
      }
    },
    subscribe(fn) {
      listeners.add(fn);
      return () => listeners.delete(fn);
    }
  };
}
```

**设计要点**：
- 纯函数，无 DOM 依赖，可在 Node 测试
- `batch()` 吸收 SSE 突发写入（替代 play 的 `createRefreshCoalescer`）
- `subscribe` 返回 unsubscribe，方便清理
- listener 抛错不中断其他 listener（容错）

### 4.4 兼容桥策略

```js
// src/state/index.js
import { createStore } from "./create-store.js";
import { createUserStore } from "./user-store.js";
import { createWorldStore } from "./world-store.js";
// ... 其余 6 个 shard

export const userStore = createUserStore(createStore);
export const worldStore = createWorldStore(createStore);
// ... 其余 6 个

/**
 * Compatibility bridge: un-migrated consumers read window.zhimuState.xxx.
 * Getter aggregates all shards, so zhimuState stays in sync with shard.set().
 *
 * Removed in Phase 4 after all 43 consumers migrate to `import { userStore }`.
 */
window.zhimuState = new Proxy({}, {
  get(_target, key) {
    if (key in userStore.get()) return userStore.get()[key];
    if (key in worldStore.get()) return worldStore.get()[key];
    // ... 其余 6 个 shard
    return undefined;
  },
  set(_target, key, value) {
    // Route write to the correct shard
    if (key in userStore.get()) userStore.set({ [key]: value });
    else if (key in worldStore.get()) worldStore.set({ [key]: value });
    // ... 其余 6 个 shard
    return true;
  }
});
```

**关键**：Proxy 让现有 43 个 consumers 无感继续工作，shard.set() 自动同步到 zhimuState，反之亦然。Phase 4 删除 Proxy。

## 六、整体回滚机制

### Commit 粒度

每个阶段、每个 shard 独立 commit，commit message 前缀：

```
A4-1: security 三端去重
A2-2: createStore 基础设施
A2-3.1: ui-store 分片迁移
A2-3.2: user-store 分片迁移
...
A2-4: 消费者迁移 + 删 zhimuState 桥
A1-收尾: 删 dom.js hack + 修复失败测试
A4-5: api-fetch 抽取 + Vite alias
```

### 回滚命令

**单阶段回滚**（以阶段 3.5 asset-store 为例）：

```powershell
git log --oneline -10                          # 找到 asset-store commit hash
git revert <commit-hash>                       # 反向 commit，保留历史
npm run check:modules && npm run build         # 验证回滚后状态
```

**整体回滚到项目开始前**：

```powershell
# 假设项目开始前的最后一个 commit 是 <pre-a1a2a4-hash>
git log --oneline -30                          # 定位起点
git reset --hard <pre-a1a2a4-hash>             # ⚠️ 会丢失未提交改动，慎用
# 或更安全：
git revert <pre-a1a2a4-hash>..HEAD             # 反向所有 commit
```

### 安全网

- **Proxy 兼容桥**：A2 阶段 2-3 期间，window.zhimuState 通过 Proxy 代理各 shard，任何 shard 迁移失败都不会断现有 consumers
- **shard 独立性**：8 个 shard 互不依赖，单个 shard 可单独 revert
- **雏形保留**：runtime-store/workspace-store/context-coordinator 在 A2 期间保留 window 桥接，最后才删

## 七、风险与对策

| 风险 | 概率 | 影响 | 对策 |
|---|---|---|---|
| Proxy 兼容桥性能 | 低 | 中 | 仅过渡期使用，Phase 4 删除；监测 render 耗时 |
